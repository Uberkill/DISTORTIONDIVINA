export default function DivinationSystem() {
  const [slots, setSlots] = useState([null, null, null, null, null]);
  const [activeSlotIndex, setActiveSlotIndex] = useState(0); 
  const [searchQuery, setSearchQuery] = useState("");
  const [isComplete, setIsComplete] = useState(false);
  const [viewMode, setViewMode] = useState("story"); // 'story' or 'data'
  const scrollRef = useRef(null);

  // Filter logic
  const filteredCards = CARD_DATABASE.filter(card => 
    card.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
    card.character.toLowerCase().includes(searchQuery.toLowerCase())
  );

  // Auto-scroll when completing
  useEffect(() => {
    if (isComplete && scrollRef.current) {
      scrollRef.current.scrollIntoView({ behavior: 'smooth' });
    }
  }, [isComplete]);

  const handleCardSelect = (card) => {
    const newSlots = [...slots];
    newSlots[activeSlotIndex] = card;
    setSlots(newSlots);
    setSearchQuery("");

    // Advance to next empty slot or finish
    if (activeSlotIndex < 4) {
      setActiveSlotIndex(activeSlotIndex + 1);
    } else {
      setIsComplete(true);
      setActiveSlotIndex(-1); // No active slot
    }
  };

  const handleSlotClick = (index) => {
    setActiveSlotIndex(index);
    setIsComplete(false);
    setSearchQuery("");
  };

  const resetSystem = () => {
    setSlots([null, null, null, null, null]);
    setActiveSlotIndex(0);
    setIsComplete(false);
    setSearchQuery("");
  };

  // --- STORY GENERATION HELPERS ---
  const getTransition = (index) => {
    const transitions = [
      "To begin, your conscious mind manifests as",
      "Beneath this intellectual surface, your heart secretly harbors",
      "This internal conflict drives your hands to act with",
      "However, lurking in your shadow is",
      "Ultimately, when all layers are stripped away, your true soul is revealed as"
    ];
    return transitions[index];
  };

  return (
    <div className="min-h-screen bg-black text-amber-500 font-mono flex flex-col relative overflow-hidden selection:bg-amber-900 selection:text-white">
      
      {/* BACKGROUND GRID & SCANLINES */}
      <div className="fixed inset-0 pointer-events-none z-0">
         <div className="absolute inset-0 bg-[linear-gradient(rgba(18,18,18,0)_50%,rgba(0,0,0,0.25)_50%),linear-gradient(90deg,rgba(255,0,0,0.06)_1px,transparent_1px),linear-gradient(rgba(255,0,0,0.06)_1px,transparent_1px)] bg-[length:100%_2px,20px_20px,20px_20px]"></div>
         <div className="absolute inset-0 bg-black/80"></div>
      </div>

      {/* TOP SYSTEM BAR */}
      <div className="relative z-10 border-b border-amber-900/50 bg-black/90 p-2 flex justify-between items-center text-xs tracking-widest">
        <div className="flex items-center gap-4">
          <span className="bg-amber-600/20 text-amber-500 px-2 py-1 border border-amber-600/50">DISTORTION_OS // CONNECTED</span>
          <span className="hidden md:inline text-zinc-600">ARCHIVE_08.EXE</span>
        </div>
        <div className="flex gap-4 text-zinc-500">
          <span>VIEW: {viewMode.toUpperCase()}</span>
          <span className="text-amber-500">BAT: 100%</span>
        </div>
      </div>

      {/* MAIN CONTENT */}
      <main className="relative z-10 flex-1 flex flex-col max-w-7xl mx-auto w-full p-4 md:p-8 gap-8">
        
        {/* HEADER */}
        <div className="flex flex-col md:flex-row justify-between items-end border-b-2 border-amber-600 pb-4">
          <div>
            <h1 className="text-4xl md:text-6xl font-bold uppercase tracking-tighter text-amber-500 drop-shadow-[0_0_10px_rgba(245,158,11,0.5)]">
              First Light
            </h1>
            <p className="text-sm text-amber-700 mt-2 max-w-xl">
              &gt;&gt; DISTRIBUTION PROTOCOL: Select 5 archetypes.
              The system requires your manual resonance input.
            </p>
          </div>
          <div className="text-right hidden md:block">
            <div className="text-xs text-amber-800">SYSTEM STATUS</div>
            <div className="text-xl text-amber-500 animate-pulse">
              {isComplete ? "CALCULATION COMPLETE" : "AWAITING INPUT..."}
            </div>
          </div>
        </div>

        {/* THE SPREAD (PROGRESS BAR VISUAL) */}
        <div className="grid grid-cols-5 gap-2 md:gap-4 w-full h-32 md:h-48 mb-8">
          {slots.map((card, index) => (
            <div 
              key={index}
              onClick={() => handleSlotClick(index)}
              className={`
                relative border-2 flex flex-col justify-between p-2 cursor-pointer transition-all duration-300 group
                ${activeSlotIndex === index ? 'border-amber-400 bg-amber-900/20 shadow-[0_0_15px_rgba(245,158,11,0.3)]' : 'border-zinc-800 bg-zinc-900/50 hover:border-zinc-600'}
                ${card ? 'border-amber-700 bg-amber-950/40' : ''}
              `}
            >
              {/* Slot Number */}
              <div className="flex justify-between items-start">
                <span className={`text-xs font-bold ${activeSlotIndex === index ? 'text-white' : 'text-zinc-600'}`}>
                  {POSITIONS[index].title}
                </span>
                {card && <div className="w-2 h-2 bg-amber-500 rounded-full shadow-[0_0_5px_rgba(245,158,11,1)]"></div>}
              </div>

              {/* Card Content or Placeholder */}
              <div className="flex-1 flex items-center justify-center text-center">
                 {card ? (
                   <div className="space-y-1 animate-in zoom-in duration-300">
                     <div className="text-lg md:text-2xl font-bold leading-none">{card.character}</div>
                     <div className="text-[10px] uppercase text-amber-500/70 tracking-wider">{card.name}</div>
                   </div>
                 ) : (
                   <div className={`text-4xl ${activeSlotIndex === index ? 'text-amber-500/20' : 'text-zinc-800'}`}>
                      +
                   </div>
                 )}
              </div>

              {/* Slot Desc */}
              <div className="text-[10px] text-center uppercase tracking-widest text-zinc-500 group-hover:text-amber-500/50">
                [{POSITIONS[index].desc}]
              </div>
              
              {/* Active Indicator Corner */}
              {activeSlotIndex === index && (
                <div className="absolute top-0 right-0 w-3 h-3 border-t-2 border-r-2 border-amber-500"></div>
              )}
              {activeSlotIndex === index && (
                <div className="absolute bottom-0 left-0 w-3 h-3 border-b-2 border-l-2 border-amber-500"></div>
              )}
            </div>
          ))}
        </div>

        {/* INPUT AREA (Only if not complete) */}
        {!isComplete && (
          <div className="flex-1 flex flex-col max-w-4xl mx-auto w-full gap-8">
             
             {/* RITUAL GUIDANCE BOX */}
             <div className="border border-amber-700/50 bg-amber-950/20 p-6 relative overflow-hidden">
                <div className="absolute top-0 left-0 bg-amber-600 text-black text-[10px] font-bold px-2 py-1">
                  GUIDANCE PROTOCOL
                </div>
                <div className="flex gap-4 items-start mt-2">
                  <Eye className="w-6 h-6 text-amber-500 mt-1 shrink-0" />
                  <div>
                    <h3 className="text-lg font-bold text-amber-100 mb-2">
                      Focus on: {POSITIONS[activeSlotIndex].title} ({POSITIONS[activeSlotIndex].desc})
                    </h3>
                    <p className="text-amber-500/80 leading-relaxed font-serif italic text-lg">
                      "{POSITIONS[activeSlotIndex].guidance}"
                    </p>
                  </div>
                </div>
             </div>

             <div className="flex flex-col gap-4">
                 <div className="text-amber-500 text-sm blink-cursor mb-2">
                   {`> SYSTEM: Awaiting manual entry for [ SLOT_${activeSlotIndex + 1} ]`}
                 </div>
                 
                 <div className="relative">
                    <div className="absolute inset-y-0 left-0 pl-4 flex items-center pointer-events-none">
                      <Terminal className="h-5 w-5 text-amber-700" />
                    </div>
                    <input
                      type="text"
                      autoFocus
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      className="w-full bg-black border border-amber-800 text-amber-100 py-4 pl-12 pr-4 focus:outline-none focus:border-amber-400 focus:ring-1 focus:ring-amber-400 placeholder-amber-900/50 font-mono text-lg uppercase"
                      placeholder="TYPE CARD NAME..."
                    />
                 </div>

                 {/* Cards Grid List */}
                 <div className="flex-1 border border-zinc-800 bg-zinc-900/30 overflow-y-auto max-h-[300px] custom-scrollbar p-2 grid grid-cols-1 md:grid-cols-2 gap-2">
                    {filteredCards.length > 0 ? (
                      filteredCards.map(card => (
                        <button
                          key={card.id}
                          onClick={() => handleCardSelect(card)}
                          className="text-left p-3 border border-transparent hover:border-amber-700 hover:bg-amber-900/20 flex items-center justify-between group transition-all"
                        >
                           <div>
                             <div className="text-amber-100 font-bold group-hover:text-amber-400">{card.character}</div>
                             <div className="text-xs text-zinc-500 group-hover:text-amber-600">{card.name}</div>
                           </div>
                           <ChevronRight className="w-4 h-4 text-zinc-700 group-hover:text-amber-500" />
                        </button>
                      ))
                    ) : (
                      <div className="p-4 text-zinc-500 italic col-span-2 text-center">NO ARCHETYPE FOUND</div>
                    )}
                 </div>
             </div>
          </div>
        )}

        {/* RESULTS AREA (Only when complete) */}
        {isComplete && (
          <div ref={scrollRef} className="animate-in slide-in-from-bottom-10 fade-in duration-700 border-t border-amber-900/30 pt-8">
            <div className="flex justify-between items-center mb-8">
               <div className="flex items-center gap-2">
                 <Sparkles className="text-amber-400 w-5 h-5" />
                 <h2 className="text-2xl font-bold text-amber-100 uppercase">Analysis Complete</h2>
               </div>
               
               {/* View Toggle */}
               <div className="flex bg-zinc-900 rounded p-1 border border-zinc-800">
                  <button 
                    onClick={() => setViewMode("story")}
                    className={`px-3 py-1 text-xs font-bold uppercase rounded flex items-center gap-2 ${viewMode === "story" ? "bg-amber-700 text-white" : "text-zinc-500 hover:text-amber-500"}`}
                  >
                    <BookOpen className="w-3 h-3" /> Narrative
                  </button>
                  <button 
                    onClick={() => setViewMode("data")}
                    className={`px-3 py-1 text-xs font-bold uppercase rounded flex items-center gap-2 ${viewMode === "data" ? "bg-amber-700 text-white" : "text-zinc-500 hover:text-amber-500"}`}
                  >
                    <Grid className="w-3 h-3" /> Data
                  </button>
               </div>
            </div>
            
            {/* STORY MODE VIEW */}
            {viewMode === "story" && (
              <div className="max-w-3xl mx-auto space-y-8">
                 <div className="bg-zinc-900/50 border border-amber-900/30 p-8 md:p-12 relative shadow-2xl">
                    <div className="absolute top-0 left-0 w-full h-1 bg-gradient-to-r from-amber-900 via-amber-500 to-amber-900"></div>
                    
                    {slots.map((card, index) => (
                      <div key={index} className="mb-6 last:mb-0">
                         <span className="text-amber-600 text-sm font-bold uppercase tracking-wider mr-2 block mb-2 md:inline md:mb-0">
                           {getTransition(index)} <span className="text-amber-200 border-b border-amber-500/30">{card.character}</span>:
                         </span>
                         <span className="text-zinc-300 font-serif text-lg leading-relaxed">
                            {card.readings[index]}
                         </span>
                      </div>
                    ))}

                    <div className="mt-12 pt-8 border-t border-amber-900/30 flex gap-4">
                      <div className="w-1 bg-amber-500 h-full"></div>
                      <div>
                        <h4 className="text-amber-500 font-bold uppercase text-sm mb-2">Final Resonance</h4>
                        <p className="text-amber-100/80 italic">
                           "The path is clear. The conflict between {slots[1]?.name} in your Heart and {slots[3]?.name} in your Shadow suggests a hidden struggle. Do not let your {slots[0]?.keywords[0]} blind you to the truth of {slots[4]?.keywords[0]}."
                        </p>
                      </div>
                    </div>
                 </div>
              </div>
            )}

            {/* DATA MODE VIEW (Original Layout) */}
            {viewMode === "data" && (
              <div className="space-y-6">
                {slots.map((card, index) => (
                  <div key={index} className="flex flex-col md:flex-row gap-4 md:gap-8 border-l-2 border-amber-800 pl-4 md:pl-8 py-2 relative hover:bg-zinc-900/30 transition-colors">
                     <div className="md:w-48 shrink-0">
                        <div className="text-xs text-amber-600 uppercase mb-1">{POSITIONS[index].title} // {POSITIONS[index].desc}</div>
                        <div className="text-xl text-amber-200 font-bold">{card.character}</div>
                        <div className="text-xs text-zinc-500">{card.name}</div>
                     </div>
                     <div className="flex-1">
                        <div className="text-lg text-amber-100/90 font-serif italic">
                          "{card.readings[index]}"
                        </div>
                     </div>
                  </div>
                ))}
              </div>
            )}
            
            <div className="mt-12 text-center">
               <button 
                 onClick={resetSystem}
                 className="px-8 py-3 border border-amber-600 text-amber-500 font-bold uppercase hover:bg-amber-900/20 transition-colors flex items-center gap-2 mx-auto"
               >
                 <RotateCcw className="w-4 h-4" /> Reset Sequence
               </button>
            </div>
          </div>
        )}

      </main>

      {/* FOOTER */}
      <footer className="relative z-10 border-t border-amber-900/50 p-2 text-[10px] text-amber-800 flex justify-between uppercase tracking-wider bg-black">
        <span>Project Moon Fan Tool // Non-Official</span>
        <span>Divination_Engine_v0.9.3</span>
      </footer>

      <style>{`
        .custom-scrollbar::-webkit-scrollbar {
          width: 8px;
        }
        .custom-scrollbar::-webkit-scrollbar-track {
          background: #18181b;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb {
          background: #451a03;
          border: 1px solid #78350f;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover {
          background: #78350f;
        }
        .blink-cursor::after {
          content: '_';
          animation: blink 1s step-end infinite;
        }
        @keyframes blink {
          0%, 100% { opacity: 1; }
          50% { opacity: 0; }
        }
      `}</style>
    </div>
  );
}
